```python
"""Wayland wire protocol codec + connection — pure stdlib.

Wire format (wayland.xml, native byte order):
  * 8-byte header: u32 object_id, u32 (length << 16) | opcode;
    length includes the header.
  * Args are 4-byte units, no extra alignment:
      u/i -> 4 bytes, f -> 4 bytes (wl_fixed 24.8),
      s -> u32 length (incl. NUL) + bytes + pad to 4,
      o/n -> 4 bytes (object / new id),
      h -> fd via SCM_RIGHTS, ZERO payload bytes.
Unknown events are skipped by header length (safe: every message is framed).
"""

import os
import socket
import struct
import time

HEADER = struct.Struct("=II")

# interface -> request table: name -> (opcode, signature)
DISPLAY_REQ = {"sync": (0, "n"), "get_registry": (1, "n")}
DISPLAY_EVT = {"error": (0, "ousu"), "delete_id": (1, "u")}
REGISTRY_REQ = {"bind": (0, "usun")}
REGISTRY_EVT = {"global": (0, "usu"), "global_remove": (1, "u")}
CALLBACK_EVT = {"done": (0, "u")}
SEAT_REQ = {"get_keyboard": (0, "n"), "get_pointer": (1, "n"),
            "get_touch": (2, "n"), "release": (3, "")}
SEAT_EVT = {"capabilities": (0, "u"), "name": (1, "s")}
KEYBOARD_EVT = {"keymap": (0, "uhu")}  # enter/leave/key/modifiers: unhandled -> skipped
OUTPUT_REQ = {"release": (0, "")}
VIRTUAL_POINTER_MGR_REQ = {
    "create_virtual_pointer": (0, "on"),
    "create_virtual_pointer_with_output": (1, "oon"),
}
VIRTUAL_POINTER_REQ = {
    "motion": (0, "uff"), "motion_absolute": (1, "uff"),
    "button": (2, "uuu"), "axis": (3, "uuf"), "frame": (4, ""),
    "axis_source": (5, "u"), "axis_stop": (6, "uu"),
    "axis_discrete": (7, "uufi"), "release": (8, ""),
}
VIRTUAL_KEYBOARD_MGR_REQ = {"create_virtual_keyboard": (0, "on")}
VIRTUAL_KEYBOARD_REQ = {
    "keymap": (0, "uhu"), "key": (1, "uuu"),
    "modifiers": (2, "uuuuu"), "release": (3, ""),
}

REQ_TABLES = {
    "wl_display": DISPLAY_REQ, "wl_registry": REGISTRY_REQ,
    "wl_seat": SEAT_REQ, "wl_output": OUTPUT_REQ,
    "zwlr_virtual_pointer_manager_v1": VIRTUAL_POINTER_MGR_REQ,
    "zwlr_virtual_pointer_v1": VIRTUAL_POINTER_REQ,
    "zwlr_virtual_keyboard_manager_v1": VIRTUAL_KEYBOARD_MGR_REQ,
    "zwlr_virtual_keyboard_v1": VIRTUAL_KEYBOARD_REQ,
}

EVT_TABLES = {
    "wl_display": DISPLAY_EVT, "wl_registry": REGISTRY_EVT,
    "wl_callback": CALLBACK_EVT, "wl_seat": SEAT_EVT,
    "wl_keyboard": KEYBOARD_EVT,
}


class WaylandError(Exception):
    """Wayland connection-level failure."""


def to_fixed(value: float) -> int:
    """float -> wl_fixed 24.8."""
    return int(value * 256.0)


def from_fixed(value: int) -> float:
    return value / 256.0


def pack_message(object_id: int, opcode: int, sig: str, args=()) -> bytes:
    payload = bytearray()
    idx = 0
    for c in sig:
        if c == "u":
            payload += struct.pack("=I", args[idx])
        elif c == "i":
            payload += struct.pack("=i", args[idx])
        elif c == "f":
            payload += struct.pack("=i", args[idx])  # wl_fixed is i32
        elif c in "on":
            payload += struct.pack("=I", args[idx])
        elif c == "s":
            raw = args[idx].encode("utf-8") + b"\0"
            payload += struct.pack("=I", len(raw))
            payload += raw + b"\0" * ((4 - len(raw) % 4) % 4)
        elif c == "h":
            pass  # fd travels via SCM_RIGHTS, not the payload
        else:
            raise ValueError(f"unknown signature char {c!r}")
        idx += 1
    return HEADER.pack(object_id, ((16 + len(payload)) << 16) | opcode) \
        + bytes(payload)


def unpack_header(data: bytes) -> tuple[int, int, int]:
    if len(data) < 8:
        raise ValueError("short Wayland header")
    obj_id, size_opcode = HEADER.unpack(data[:8])
    return obj_id, size_opcode & 0xFFFF, size_opcode >> 16


def unpack_args(sig: str, data: bytes, offset: int, fds: list) -> tuple[list, int]:
    args = []
    p = offset
    for c in sig:
        if c == "u":
            args.append(struct.unpack_from("=I", data, p)[0]); p += 4
        elif c == "i":
            args.append(struct.unpack_from("=i", data, p)[0]); p += 4
        elif c == "f":
            args.append(struct.unpack_from("=i", data, p)[0]); p += 4
        elif c in "on":
            args.append(struct.unpack_from("=I", data, p)[0]); p += 4
        elif c == "s":
            (slen,) = struct.unpack_from("=I", data, p); p += 4
            raw = data[p:p + slen]
            args.append(raw.rstrip(b"\0").decode("utf-8", errors="replace"))
            p += (slen + 3) & ~3
        elif c == "h":
            args.append(fds.pop(0) if fds else -1)
        else:
            raise ValueError(f"unknown signature char {c!r}")
    return args, p


class WaylandConnection:
    """One Wayland display connection: send/recv + event dispatch.

    Events dispatch to `_on_<iface>_<name>` methods on this object, or —
    if absent — on `self.handler` (the session that owns the connection).
    """

    def __init__(self, fd: int):
        self._sock = socket.socket(fileno=fd)
        self._sock.setblocking(False)
        self._buf = bytearray()
        self._pending_fds: list = []
        self.objects: dict[int, str] = {}  # object id -> interface name
        self.handler = None
        self._next_id = 1
        self.display_id = self.alloc_id()
        self.objects[self.display_id] = "wl_display"

    @classmethod
    def connect(cls, display: str) -> "WaylandConnection":
        runtime = os.environ.get("XDG_RUNTIME_DIR")
        if not runtime:
            raise WaylandError("XDG_RUNTIME_DIR is not set")
        path = os.path.join(runtime, display)
        sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        sock.settimeout(5.0)
        try:
            sock.connect(path)
        except OSError as e:
            sock.close()
            raise WaylandError(f"cannot connect to {path}: {e}") from e
        return cls(sock.detach())

    @property
    def fd(self) -> int:
        return self._sock.fileno()

    def alloc_id(self) -> int:
        obj_id = self._next_id
        self._next_id += 1
        return obj_id

    def send(self, obj_id: int, iface: str, name: str, args) -> None:
        opcode, sig = REQ_TABLES[iface][name]
        payload = pack_message(obj_id, opcode, sig, args)
        fds = [a for a, c in zip(args, sig)
               if c == "h" and isinstance(a, int) and a >= 0]
        try:
            if fds:
                self._sock.sendmsg(
                    [payload], [(socket.SOL_SOCKET, socket.SCM_RIGHTS,
                                 struct.pack(f"{len(fds)}i", *fds))])
            else:
                self._sock.sendall(payload)
        except OSError as e:
            raise WaylandError(f"Wayland send failed: {e}") from e

    def bind(self, registry_id: int, name: int, iface: str,
             version: int) -> int:
        obj_id = self.alloc_id()
        self.objects[obj_id] = iface
        self.send(registry_id, "wl_registry", "bind",
                  (name, iface, version, obj_id))
        return obj_id

    def sync(self, timeout: float = 2.0) -> None:
        """Roundtrip: sync + wait for the callback's done event."""
        callback_id = self.alloc_id()
        self.objects[callback_id] = "wl_callback"
        self.send(self.display_id, "wl_display", "sync", (callback_id,))
        deadline = time.monotonic() + timeout
        while callback_id in self.objects:
            self.pump(deadline - time.monotonic())
            if time.monotonic() >= deadline:
                raise WaylandError("wl_display.sync roundtrip timed out")

    def close(self) -> None:
        try:
            self._sock.close()
        except OSError:
            pass

    def pump(self, timeout: float = 0.0) -> None:
        """Drain pending events (non-blocking; timeout is a bound only)."""
        try:
            data, ancdata, _flags, _addr = self._sock.recvmsg(
                65536, socket.CMSG_SPACE(16 * 4))
        except BlockingIOError:
            return
        except OSError as e:
            raise WaylandError(f"Wayland recv failed: {e}") from e
        if not data:
            raise WaylandError("Wayland server closed the connection")
        for level, ctype, cdata in ancdata:
            if level == socket.SOL_SOCKET and ctype == socket.SCM_RIGHTS:
                self._pending_fds.extend(
                    struct.unpack(f"{len(cdata) // 4}i", cdata))
        self._buf.extend(data)
        while len(self._buf) >= 8:
            obj_id, size, opcode = unpack_header(bytes(self._buf[:8]))
            if len(self._buf) < size:
                break
            payload = bytes(self._buf[8:size])
            del self._buf[:size]
            self._dispatch(obj_id, opcode, payload)

    def _dispatch(self, obj_id: int, opcode: int, payload: bytes) -> None:
        iface = self.objects.get(obj_id, "")
        evts = EVT_TABLES.get(iface, {})
        name = sig = None
        for evt_name, (evt_opcode, evt_sig) in evts.items():
            if evt_opcode == opcode:
                name, sig = evt_name, evt_sig
                break
        if name is None:
            return  # unknown/unregistered event — framed, safe to skip
        args, _ = unpack_args(sig, payload, 0, self._pending_fds)
        handler = getattr(self, f"_on_{iface}_{name}", None)
        if handler is None and self.handler is not None:
            handler = getattr(self.handler, f"_on_{iface}_{name}", None)
        if handler is not None:
            handler(obj_id, args)

    def _on_wl_callback_done(self, obj_id, args) -> None:
        self.objects.pop(obj_id, None)  # resolves pending sync()
```

```python
        (5, "zwlr_virtual_keyboard_manager_v1", 1),
    ]

    def __init__(self, globals=None, keyboard=True, output=True, path=None):
        self.globals = list(globals) if globals else list(self.GLOBALS)
        self.keyboard = keyboard
        self.output = output
        self.path = path
        self.global_names = {
            giface: gname for gname, giface, _gver in self.globals}
        self.received = []          # (obj_id, iface, name, args)
        self.bound = {}             # new object id -> interface name
        self.keymaps_sent = []      # bytes received on zwlr virtual keyboards
        if path is not None:
            self.server_sock = socket.socket(socket.AF_UNIX,
                                             socket.SOCK_STREAM)
            self.server_sock.bind(path)
            self.server_sock.listen(1)
            self.client_sock = None
        else:
            self.client_sock, self.server_sock = socket.socketpair()
        self._thread = threading.Thread(target=self._serve, daemon=True)

    def start(self):
        self._thread.start()

    def close(self):
        try:
            self.server_sock.close()
        except OSError:
            pass
        if self.path is not None:
            try:
                os.unlink(self.path)
            except OSError:
                pass

    def join(self, timeout=3.0):
        self._thread.join(timeout)

    def client_fd(self):
        assert self.client_sock is not None
        return self.client_sock.detach()

    def messages(self, iface, name=None):
        return [m for m in self.received
                if m[1] == iface and (name is None or m[2] == name)]

    def wait_messages(self, count, timeout=3.0, iface=None, name=None):
        deadline = time.monotonic() + timeout
        while time.monotonic() < deadline:
            got = [m for m in self.received
                   if (iface is None or m[1] == iface)
                   and (name is None or m[2] == name)]
            if len(got) >= count:
                return got
            time.sleep(0.01)
        raise AssertionError(
            f"expected {count} messages (iface={iface}, name={name}), "
            f"got {len(got)}")

    def _send(self, obj_id, opcode, sig, args, fds=()):
        payload = w.pack_message(obj_id, opcode, sig, args)
        if fds:
            self.server_sock.sendmsg(
                [payload],
                [(socket.SOL_SOCKET, socket.SCM_RIGHTS,
                  struct.pack(f"{len(fds)}i", *fds))])
        else:
            self.server_sock.sendall(payload)

    def _serve(self):
        self._buf = bytearray()
        try:
            while True:
                data, ancdata, _flags, _addr = self.server_sock.recvmsg(
                    65536, socket.CMSG_SPACE(16 * 4))
                if not data:
                    break
                pending_fds = []
                for level, ctype, cdata in ancdata:
                    if level == socket.SOL_SOCKET and ctype == socket.SCM_RIGHTS:
                        pending_fds.extend(
                            struct.unpack(f"{len(cdata) // 4}i", cdata))
                self._buf.extend(data)
                while len(self._buf) >= 8:
                    obj_id, size, opcode = w.unpack_header(
                        bytes(self._buf[:8]))
                    if len(self._buf) < size:
                        break
                    payload = bytes(self._buf[8:size])
                    del self._buf[:size]
                    self._handle(obj_id, opcode, payload, pending_fds)
        except OSError:
            pass  # client closed

    def _handle(self, obj_id, opcode, payload, pending_fds):
        iface = self.bound.get(obj_id, "wl_display")
        table = w.REQ_TABLES.get(iface, {})
        name = sig = None
        for req_name, (req_opcode, req_sig) in table.items():
            if req_opcode == opcode:
                name, sig = req_name, req_sig
                break
        if name is None:
            return
        args, _ = w.unpack_args(sig, payload, 0, pending_fds)
        self.received.append((obj_id, iface, name, args))

        if iface == "wl_display" and name == "get_registry":
            registry_id = args[0]
            self.bound[registry_id] = "wl_registry"
            for gname, giface, gver in self.globals:
                self._send(registry_id, *w.REGISTRY_EVT["global"],
                           (gname, giface, gver))
        elif iface == "wl_display" and name == "sync":
            cb_id = args[0]
            self._send(cb_id, *w.CALLBACK_EVT["done"], (0,))
        elif iface == "wl_registry" and name == "bind":
            _gname, iface_name, _version, new_id = args
            self.bound[new_id] = iface_name
            if iface_name == "wl_seat" and self.keyboard:
                self._send(new_id, *w.SEAT_EVT["capabilities"], (2,))
        elif iface == "wl_seat" and name == "get_keyboard":
            kb_id = args[0]
            self.bound[kb_id] = "wl_keyboard"
            keymap_fd = os.memfd_create("keymap")
            os.write(keymap_fd, self.KEYMAP)
            os.lseek(keymap_fd, 0, os.SEEK_SET)
            self._send(kb_id, *w.KEYBOARD_EVT["keymap"],
                       (1, -1, len(self.KEYMAP)), fds=[keymap_fd])
            os.close(keymap_fd)
        elif iface == "zwlr_virtual_pointer_manager_v1" and name in (
                "create_virtual_pointer", "create_virtual_pointer_with_output"):
            new_id = args[-1]
            self.bound[new_id] = "zwlr_virtual_pointer_v1"
        elif iface == "zwlr_virtual_keyboard_manager_v1" \
                and name == "create_virtual_keyboard":
            new_id = args[-1]
            self.bound[new_id] = "zwlr_virtual_keyboard_v1"
        elif iface == "zwlr_virtual_keyboard_v1" and name == "keymap":
            _format, keymap_fd, _size = args
            data = b""
            while True:
                chunk = os.read(keymap_fd, 65536)
                if not chunk:
                    break
                data += chunk
            os.close(keymap_fd)
            self.keymaps_sent.append(data)
```

```python
"""wlroots zwlr virtual input backend (sway / river / labwc / wayfire).

Pure stdlib Wayland client.  No token, no dialog: on wlroots the zwlr
protocols are compositor-native.  The keyboard chain reuses xkb.py with
the SEAT's real keymap (read from wl_keyboard.keymap), which is also
forwarded to the virtual keyboard.

zwlr timestamps are u32 milliseconds (CLOCK_MONOTONIC-relative).
"""

import logging
import os
import time

from .base import (
    AXIS_HORIZONTAL,
    AXIS_VERTICAL,
    BUTTONS,
    PRESS,
    RELEASE,
    Backend,
    BackendError,
    SessionNotStartedError,
    UnsupportedCharacterError,
)
from .wayland import WaylandConnection, WaylandError, from_fixed, to_fixed
from ..keysyms import lookup as lookup_keysym
from .xkb import build_resolver

log = logging.getLogger(__name__)

KEYBOARD_CAP = 2
KEYMAP_FORMAT_XKB_V1 = 1
AXIS_SOURCE_WHEEL = 0


class WlrootsSession(Backend):
    """Backend over zwlr_virtual_pointer_v1 / zwlr_virtual_keyboard_v1."""

    name = "wlroots"

    def __init__(self, display: str | None = None,
                 conn: WaylandConnection | None = None):
        self.display = display
        self._conn = conn
        self._state = "init"
        self._last_error: BackendError | None = None
        self.devices: list = []
        # registry state
        self._registry_id: int | None = None
        self._seat_name: int | None = None
        self._output_name: int | None = None
        self._vp_manager: int | None = None
        self._vk_manager: int | None = None
        # bound objects
        self._seat_id: int | None = None
        self._keyboard_id: int | None = None
        self._vp_id: int | None = None
        self._vk_id: int | None = None
        # keyboard state
        self._seen_keyboard_cap = False
        self._keymap_text: str | None = None
        self._resolver = None

    # -- state -----------------------------------------------------------

    @property
    def state(self) -> str:
        return self._state

    @property
    def last_error(self) -> BackendError | None:
        return self._last_error

    def start(self) -> None:
        if self._state in ("starting", "started"):
            return
        self._state = "starting"
        self._last_error = None
        self.devices = []
        try:
            display = self.display or os.environ.get("WAYLAND_DISPLAY")
            if not display:
                raise BackendError("WAYLAND_DISPLAY is not set — cannot "
                                   "reach a wlroots compositor")
            conn = self._conn or WaylandConnection.connect(display)
            conn.handler = self
            self._conn = conn
            self._collect_globals(conn)
            self._check_required_globals()
            self._bind_seat_and_keymap(conn)
            self._create_devices(conn)
        except (BackendError, WaylandError, OSError) as e:
            self._fail(e if isinstance(e, BackendError) else BackendError(str(e)))
            return
        self._state = "started"
        log.info("wlroots session started; devices=%s", self.devices)

    def _collect_globals(self, conn) -> None:
        self._registry_id = conn.alloc_id()
        conn.objects[self._registry_id] = "wl_registry"
        conn.send(conn.display_id, "wl_display", "get_registry",
                  (self._registry_id,))
        conn.sync()  # globals have all arrived by the roundtrip's end

    def _check_required_globals(self) -> None:
        missing = [name for name, value in (
            ("wl_seat", self._seat_name),
            ("zwlr_virtual_pointer_manager_v1", self._vp_manager),
            ("zwlr_virtual_keyboard_manager_v1", self._vk_manager),
        ) if value is None]
        if missing:
            raise BackendError(
                "compositor does not provide required globals: "
                + ", ".join(missing))

    def _bind_seat_and_keymap(self, conn) -> None:
        if self._seat_name is not None:
            self._seat_id = conn.bind(self._registry_id,
                                      self._seat_name, "wl_seat", 5)
            conn.sync()  # capabilities arrive here
            if not self._seen_keyboard_cap:
                log.warning("seat has no keyboard capability; "
                            "keyboard input disabled")
            else:
                self._keyboard_id = conn.alloc_id()
                conn.objects[self._keyboard_id] = "wl_keyboard"
                conn.send(self._seat_id, "wl_seat", "get_keyboard",
                          (self._keyboard_id,))
                conn.sync()  # keymap arrives here
        if self._keymap_text is None:
            log.warning("no seat keymap received; keyboard input disabled")

    def _create_devices(self, conn) -> None:
        assert self._vp_manager is not None and self._vk_manager is not None
        self._vp_id = conn.alloc_id()
        conn.objects[self._vp_id] = "zwlr_virtual_pointer_v1"
        if self._output_name is not None:
            output_id = conn.bind(self._registry_id, self._output_name,
                                  "wl_output", 3)
            conn.send(self._vp_manager,
                      "zwlr_virtual_pointer_manager_v1",
                      "create_virtual_pointer_with_output",
                      (self._seat_id, output_id, self._vp_id))
            self.devices.append("virtual_pointer_absolute")
        else:
            conn.send(self._vp_manager,
                      "zwlr_virtual_pointer_manager_v1",
                      "create_virtual_pointer",
                      (self._seat_id, self._vp_id))
            self.devices.append("virtual_pointer")
        self._vk_id = conn.alloc_id()
        conn.objects[self._vk_id] = "zwlr_virtual_keyboard_v1"
        conn.send(self._vk_manager,
                  "zwlr_virtual_keyboard_manager_v1",
                  "create_virtual_keyboard",
                  (self._seat_id, self._vk_id))
        self.devices.append("virtual_keyboard")
        if self._keymap_text is not None:
            data = self._keymap_text.encode("utf-8")
            fd = os.memfd_create("pywaylandauto-keymap")
            os.write(fd, data)
            os.lseek(fd, 0, os.SEEK_SET)
            try:
                conn.send(self._vk_id, "zwlr_virtual_keyboard_v1",
                          "keymap", (KEYMAP_FORMAT_XKB_V1, fd, len(data)))
            finally:
                os.close(fd)  # the compositor holds its own dup

    def stop(self) -> None:
        conn = self._conn
        if conn is not None:
            try:
                if self._vp_id is not None:
                    conn.send(self._vp_id, "zwlr_virtual_pointer_v1",
                              "release", ())
                if self._vk_id is not None:
                    conn.send(self._vk_id, "zwlr_virtual_keyboard_v1",
                              "release", ())
            except (WaylandError, OSError):
                pass
            conn.close()
        self._state = "stopped"
        self._conn = None
        self.devices = []

    def _fail(self, error: BackendError) -> None:
        self._state = "stopped"
        self._last_error = error
        log.error("wlroots session failed: %s", error)

    def status(self) -> dict:
        return {
            "state": self._state,
            "devices": list(self.devices),
            "transport": "zwlr",
            "has_token": False,
            "persist_mode": None,
            "error": type(self._last_error).__name__ if self._last_error else None,
        }

    def io_fds(self) -> dict[int, str]:
        if self._conn is not None:
            return {self._conn.fd: "wayland"}
        return {}

    def pump(self, fd: int) -> None:
        if self._conn is not None:
            self._conn.pump()

    # -- registry/seat/keyboard event handlers (main-loop safe) ----------

    def _on_wl_registry_global(self, obj_id, args) -> None:
        name, iface, _version = args
        if iface == "wl_seat" and self._seat_name is None:
            self._seat_name = name
        elif iface == "wl_output" and self._output_name is None:
            self._output_name = name
        elif iface == "zwlr_virtual_pointer_manager_v1":
            self._vp_manager = name
        elif iface == "zwlr_virtual_keyboard_manager_v1":
            self._vk_manager = name

    def _on_wl_seat_capabilities(self, obj_id, args) -> None:
        if args[0] & KEYBOARD_CAP:
            self._seen_keyboard_cap = True

    def _on_wl_keyboard_keymap(self, obj_id, args) -> None:
        keymap_format, fd, size = args
        try:
            data = b""
            while len(data) < size:
                chunk = os.read(fd, size - len(data))
                if not chunk:
                    break
                data += chunk
        except OSError:
            data = b""
        finally:
            try:
                os.close(fd)
            except OSError:
                pass
        if keymap_format == KEYMAP_FORMAT_XKB_V1 and data:
            self._keymap_text = data.decode("utf-8", errors="replace")
            log.info("wlroots keymap: %d bytes of xkb text", len(data))
        else:
            log.warning("seat keymap format %d unsupported", keymap_format)

    # -- keysym resolution (shared with the EIS keyboard chain) ----------

    def _resolve_key(self, ks: int) -> tuple[int, int]:
        if self._resolver is None:
            self._resolver = build_resolver(self._keymap_text)
        resolved = self._resolver(ks)
        if resolved is None:
            raise UnsupportedCharacterError(
                f"keysym 0x{ks:04x} is not in the keymap")
        return resolved
```

```python
            from .base import PermissionPendingError
            raise PermissionPendingError("backend start in progress")
        if self._last_error is not None:
            raise self._last_error
        raise SessionNotStartedError(
            "no wlroots session — call session.start first")

    @staticmethod
    def _now_ms() -> int:
        return int(time.monotonic() * 1000)

    def move_rel(self, dx: float, dy: float) -> None:
        self._require_started()
        self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1", "motion",
                        (self._now_ms(), to_fixed(dx), to_fixed(dy)))
        self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1", "frame", ())

    def move_abs(self, x: float, y: float, translator=None) -> None:
        if translator is not None:
            x, y = translator.translate(x, y)
        self._require_started()
        if self._output_name is None:
            raise BackendError(
                "absolute pointer motion needs a wl_output (none was "
                "announced by the compositor)")
        self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1",
                        "motion_absolute",
                        (self._now_ms(), to_fixed(x), to_fixed(y)))
        self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1", "frame", ())

    @staticmethod
    def _button_code(button) -> int:
        if isinstance(button, int):
            return button
        code = BUTTONS.get(str(button).lower())
        if code is None:
            raise ValueError(f"unknown button {button!r}")
        return code

    @staticmethod
    def _press_state(state: str) -> int:
        if state == "press":
            return PRESS
        if state == "release":
            return RELEASE
        raise ValueError(f"state must be 'press' or 'release', got {state!r}")

    def button(self, button, state: str) -> None:
        code = self._button_code(button)
        press = self._press_state(state)
        self._require_started()
        self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1", "button",
                        (self._now_ms(), code, press))
        self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1", "frame", ())

    def scroll(self, dx: int = 0, dy: int = -1, discrete: bool = True) -> None:
        self._require_started()
        if discrete:
            if dy:
                self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1",
                                "axis_discrete",
                                (self._now_ms(), AXIS_VERTICAL, 0, int(dy)))
            if dx:
                self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1",
                                "axis_discrete",
                                (self._now_ms(), AXIS_HORIZONTAL, 0, int(dx)))
        else:
            self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1",
                            "axis_source", (AXIS_SOURCE_WHEEL,))
            if dy:
                self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1",
                                "axis",
                                (self._now_ms(), AXIS_VERTICAL,
                                 to_fixed(float(dy))))
            if dx:
                self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1",
                                "axis",
                                (self._now_ms(), AXIS_HORIZONTAL,
                                 to_fixed(float(dx))))
        self._conn.send(self._vp_id, "zwlr_virtual_pointer_v1", "frame", ())

    def _key(self, ks: int, press: int) -> None:
        keycode, level = self._resolve_key(ks)
        if level == 0:
            self._conn.send(self._vk_id, "zwlr_virtual_keyboard_v1", "key",
                            (self._now_ms(), keycode, press))
            return
        if level == 1:
            shift_keycode, _ = self._resolve_key(0xFFE1)  # Shift_L
            if press:
                self._conn.send(self._vk_id, "zwlr_virtual_keyboard_v1",
                                "key", (self._now_ms(), shift_keycode, PRESS))
                self._conn.send(self._vk_id, "zwlr_virtual_keyboard_v1",
                                "key", (self._now_ms(), keycode, PRESS))
            else:
                self._conn.send(self._vk_id, "zwlr_virtual_keyboard_v1",
                                "key", (self._now_ms(), keycode, RELEASE))
                self._conn.send(self._vk_id, "zwlr_virtual_keyboard_v1",
                                "key", (self._now_ms(), shift_keycode, RELEASE))
            return
        raise UnsupportedCharacterError(
            f"keysym 0x{ks:04x} needs modifier level {level} (unsupported)")

    def key(self, keysym, state: str) -> None:
        ks = lookup_keysym(keysym)
        if ks is None:
            raise ValueError(f"unknown keysym {keysym!r}")
        if state == "tap":
            self.key(ks, "press")
            self.key(ks, "release")
            return
        press = self._press_state(state)
        self._require_started()
        if self._keymap_text is None and self._keyboard_id is None:
            raise BackendError("no seat keymap — keyboard input disabled")
        self._key(ks, press)

    def type_text(self, text: str) -> None:
        self._require_started()
        if self._keymap_text is None and self._keyboard_id is None:
            raise BackendError("no seat keymap — keyboard input disabled")
        for ch in text:
            codepoint = ord(ch)
            if codepoint > 0xFF:
                raise UnsupportedCharacterError(
                    f"character {ch!r} (U+{codepoint:04X}) is beyond Latin-1")
            self._key(codepoint, PRESS)
            self._key(codepoint, RELEASE)
```

```python
"""Backend selection: auto-detect ladder + explicit override.

PYWAYLANDAUTO_BACKEND=auto (default) | portal | wlroots.

auto: probe the Wayland display for the zwlr virtual input globals;
if present use wlroots (no dialog, no token on sway et al), else portal
(the portal reports its own unavailability if missing).
"""

import logging
import os

from .base import Backend
from .portal import PortalSession
from .wayland import WaylandConnection, WaylandError
from .wlroots import WlrootsSession

log = logging.getLogger(__name__)

ZWLR_GLOBALS = (
    "zwlr_virtual_pointer_manager_v1",
    "zwlr_virtual_keyboard_manager_v1",
)


class _Probe:
    """Collects registry globals for probe_wlroots()."""

    def __init__(self):
        self.ifaces: set = set()

    def _on_wl_registry_global(self, obj_id, args):
        self.ifaces.add(args[1])


def probe_wlroots(display: str | None = None,
                  timeout: float = 2.0) -> bool:
    display = display or os.environ.get("WAYLAND_DISPLAY")
    if not display:
        return False
    try:
        conn = WaylandConnection.connect(display)
    except (WaylandError, OSError):
        return False
    try:
        probe = _Probe()
        conn.handler = probe
        registry = conn.alloc_id()
        conn.objects[registry] = "wl_registry"
        conn.send(conn.display_id, "wl_display", "get_registry", (registry,))
        try:
            conn.sync(timeout=timeout)
        except WaylandError:
            return False
        return all(iface in probe.ifaces for iface in ZWLR_GLOBALS)
    finally:
        conn.close()


def detect_backend(backend: str | None = None,
                   env: dict | None = None,
                   display: str | None = None) -> Backend:
    env = env if env is not None else os.environ
    choice = backend or env.get("PYWAYLANDAUTO_BACKEND", "auto")
    if choice == "portal":
        log.info("Backend: portal (explicit)")
        return PortalSession()
    if choice == "wlroots":
        log.info("Backend: wlroots (explicit)")
        return WlrootsSession(display=display)
    if probe_wlroots(display=display):
        log.info("Backend: wlroots (zwlr globals detected)")
        return WlrootsSession(display=display)
    log.info("Backend: portal (no wlroots protocols)")
    return PortalSession()
```

```python
        self._next_client_id += 1
        return self._next_client_id

    def sync(self, timeout: float = 2.0) -> None:
        """Roundtrip: ei_connection.sync + wait for ei_callback.done."""
        cb_id = self._alloc_object_id()
        self._objects[cb_id] = "ei_callback"
        self._last_callback_id = cb_id
        self._send(self._connection_id, *m.CONNECTION_REQ["sync"],
                   (self._last_serial, cb_id))
        deadline = time.monotonic() + timeout
        while cb_id in self._objects:
            self._pump_once()
            if time.monotonic() >= deadline:
                raise EisError("ei_connection.sync timed out")
```
